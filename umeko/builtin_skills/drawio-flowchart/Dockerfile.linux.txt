# Build from an unpacked drawio-flowchart SkillHub ZIP, at its SKILL.md root.
# docker build -f drawio-flowchart/Dockerfile.linux.txt -t umeko-drawio:1.0.2 .
FROM ubuntu:24.04
ARG DRAWIO_VERSION=32.4.1
ARG DRAWIO_SHA256=f4455b3bba3d3b539c3cd069218fc64f0462b1b01610fe8d11acc3ca8566f32b
RUN apt-get -o Acquire::Retries=3 update && apt-get -o Acquire::Retries=3 install -y --no-install-recommends \
    python3 ca-certificates curl xvfb xauth fontconfig fonts-noto-cjk \
    && curl --fail --location --retry 3 \
       "https://github.com/jgraph/drawio-desktop/releases/download/v${DRAWIO_VERSION}/drawio-amd64-${DRAWIO_VERSION}.deb" \
       --output /tmp/drawio.deb \
    && printf '%s  /tmp/drawio.deb\n' "$DRAWIO_SHA256" | sha256sum --check - \
    && apt-get -o Acquire::Retries=3 install -y --no-install-recommends /tmp/drawio.deb \
    && rm -f /tmp/drawio.deb && rm -rf /var/lib/apt/lists/* \
    && useradd --create-home --uid 10001 --shell /bin/sh umeko \
    && mkdir /work && chown umeko:umeko /work
# Electron loads ALSA even for file export; the desktop .deb does not require it.
RUN apt-get -o Acquire::Retries=3 update \
    && apt-get -o Acquire::Retries=3 install -y --no-install-recommends libasound2t64 \
    && rm -rf /var/lib/apt/lists/*
COPY . /skill
RUN sed -i 's/\r$//' /skill/drawio-flowchart/drawio_headless.sh \
    && install -m 0755 /skill/drawio-flowchart/drawio_headless.sh /usr/local/bin/drawio-headless \
    && mkdir -p /tmp/.X11-unix && chmod 1777 /tmp/.X11-unix
ENV DRAWIO_PATH=/usr/local/bin/drawio-headless
# Explicit container-only Electron setting. The process itself remains non-root.
ENV DRAWIO_NO_SANDBOX=1
USER 10001:10001
WORKDIR /work
ENTRYPOINT ["python3", "/skill/scripts/run.py"]
